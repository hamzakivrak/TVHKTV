[script]
(function () {
  if (window.__tvcAd) return; window.__tvcAd = 1;
  try { Object.defineProperty(document, 'hidden', { get: function () { return false; }, configurable: true }); Object.defineProperty(document, 'visibilityState', { get: function () { return 'visible'; }, configurable: true }); document.addEventListener('visibilitychange', function (e) { e.stopImmediatePropagation(); }, true); } catch (e) {}
  var KEYS = ['adPlacements', 'playerAds', 'adSlots', 'adBreakHeartbeatParams', 'adBreakParams'];
  function prune(o, d) {
    try {
      if (!o || typeof o !== 'object' || (d || 0) > 4) return;
      for (var i = 0; i < KEYS.length; i++) {
        if (KEYS[i] in o) { try { delete o[KEYS[i]]; } catch (e) { o[KEYS[i]] = undefined; } }
      }
      if (o.playerResponse) prune(o.playerResponse, (d || 0) + 1);
      if (o.response) prune(o.response, (d || 0) + 1);
    } catch (e) {}
  }
  var oParse = JSON.parse;
  JSON.parse = function () {
    var r = oParse.apply(this, arguments);
    try { if (r && typeof r === 'object') prune(r, 0); } catch (e) {}
    return r;
  };
  var oFetch = window.fetch;
  if (oFetch) {
    window.fetch = function () {
      var a = arguments, p = oFetch.apply(this, a);
      try {
        var u = String((a[0] && a[0].url) || a[0] || '');
        if (u.indexOf('/youtubei/v1/player') < 0 && u.indexOf('/youtubei/v1/next') < 0) return p;
        return p.then(function (res) {
          return res.clone().text().then(function (t) {
            try {
              var j = oParse(t); prune(j, 0);
              return new Response(JSON.stringify(j), { status: res.status, statusText: res.statusText, headers: res.headers });
            } catch (e) { return res; }
          }, function () { return res; });
        });
      } catch (e) { return p; }
    };
  }
  // Altyazi gorunumu: uygulama (ana sayfa) ayar gonderir, burada YouTube altyazisina uygulanir
  window.addEventListener('message', function (e) {
    try {
      var d = e.data; if (typeof d !== 'string' || d.indexOf('tvcSub') < 0) return;
      var c = JSON.parse(d).tvcSub; if (!c) return;
      var st = document.getElementById('tvc-sub-style');
      if (!st) { st = document.createElement('style'); st.id = 'tvc-sub-style'; (document.head || document.documentElement).appendChild(st); }
      st.textContent =
        '.caption-window { zoom:' + (c.size / 100) + ' !important; top:auto !important; bottom:' + c.pos + '% !important; margin-bottom:0 !important; }' +
        '.ytp-caption-segment { color:' + c.fg + ' !important; background:' + c.bg + ' !important; text-shadow:' + c.sh + ' !important; }';
    } catch (x) {}
  });
  // Yedek: reklam yine de baslarsa sessize al, hizlandir, atla dugmesine bas
  function tick() {
    try {
      var pl = document.getElementById('movie_player') || document.querySelector('.html5-video-player');
      var v = document.querySelector('video');
      if (pl && v && (pl.classList.contains('ad-showing') || pl.classList.contains('ad-interrupting'))) {
        v.muted = true; v.playbackRate = 16;
        if (isFinite(v.duration) && v.duration > 0) { try { v.currentTime = v.duration; } catch (e) {} }
        var b = document.querySelector('.ytp-ad-skip-button, .ytp-ad-skip-button-modern, .ytp-skip-ad-button');
        if (b) b.click();
        window.__tvcWasAd = 1;
      } else if (window.__tvcWasAd && v) {
        window.__tvcWasAd = 0; v.playbackRate = 1; v.muted = false;
      }
    } catch (e) {}
  }
  setInterval(tick, 300);
})();
